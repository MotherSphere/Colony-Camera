cmake_minimum_required(VERSION 3.24)
project(ColonyCamera VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_program(CARGO cargo REQUIRED)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/deps")
set(COMMONLIB "${DEPS}/commonlib")
set(SPDLOG_USE_STD_FORMAT ON CACHE BOOL "" FORCE)
set(SPDLOG_WCHAR_SUPPORT ON CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/spdlog" EXCLUDE_FROM_ALL)
include("${COMMONLIB}/cmake/sourcelist.cmake")
list(TRANSFORM SOURCES PREPEND "${COMMONLIB}/")
add_library(CommonLibSSE STATIC ${SOURCES})
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(CommonLibSSE PUBLIC /Zc:preprocessor /Zc:__cplusplus /permissive- /bigobj)
endif()
target_include_directories(CommonLibSSE PUBLIC "${COMMONLIB}/include" "${DEPS}/directxmath/Inc" "${DEPS}/directxtk/Inc")
target_compile_definitions(CommonLibSSE PUBLIC ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 HAS_SKYRIM_MULTI_TARGETING=1 WINVER=0x0601 _WIN32_WINNT=0x0601 PRIVATE COMMONLIB_VERSION="7.5.2")
target_link_libraries(CommonLibSSE PUBLIC spdlog::spdlog advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version)
target_precompile_headers(CommonLibSSE PRIVATE "${COMMONLIB}/include/SKSE/Impl/PCH.h")
set(CC_RUST_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo" CACHE PATH "Cargo artifacts directory")
set(RUST_LIB "${CC_RUST_TARGET_DIR}/x86_64-pc-windows-msvc/release/colony_camera.lib")
add_custom_command(OUTPUT "${RUST_LIB}"
    COMMAND ${CMAKE_COMMAND} -E env "RUSTFLAGS=-Ctarget-feature=+crt-static" ${CARGO} build --locked --release --target x86_64-pc-windows-msvc --target-dir "${CC_RUST_TARGET_DIR}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS src/lib.rs src/config.rs src/coordinator.rs src/first_person.rs Cargo.toml Cargo.lock VERBATIM)
add_custom_target(camera_rust DEPENDS "${RUST_LIB}")
add_library(ColonyCamera SHARED plugin/main.cpp)
add_dependencies(ColonyCamera camera_rust)
target_link_libraries(ColonyCamera PRIVATE "${RUST_LIB}" CommonLibSSE user32 shell32 ntdll userenv ws2_32 synchronization)
target_compile_options(ColonyCamera PRIVATE /utf-8 /W4)
target_link_options(ColonyCamera PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF)
# Cross-language ABI smoke test, also runnable with Wine independently of Skyrim.
add_executable(camera_abi tests/abi.cpp)
add_dependencies(camera_abi camera_rust)
target_link_libraries(camera_abi PRIVATE CommonLibSSE "${RUST_LIB}" ntdll userenv ws2_32 bcrypt synchronization)
target_compile_options(camera_abi PRIVATE /UNDEBUG)
install(TARGETS ColonyCamera RUNTIME DESTINATION SKSE/Plugins)
install(FILES assets/SKSE/Plugins/ColonyCamera.ini DESTINATION SKSE/Plugins)
add_executable(camera_load tests/load.cpp)
target_compile_options(camera_load PRIVATE /UNDEBUG)
enable_testing()
add_test(NAME camera_abi COMMAND camera_abi)
add_test(NAME camera_load COMMAND camera_load $<TARGET_FILE:ColonyCamera>)
add_executable(camera_ownership tests/ownership.cpp)
target_compile_options(camera_ownership PRIVATE /UNDEBUG)
add_test(NAME camera_ownership COMMAND camera_ownership)
add_executable(camera_config_store tests/config_store.cpp)
add_dependencies(camera_config_store camera_rust)
target_link_libraries(camera_config_store PRIVATE "${RUST_LIB}" ntdll userenv ws2_32 bcrypt synchronization)
add_test(NAME camera_config_store COMMAND camera_config_store)
add_executable(camera_hook_validation tests/hook_validation.cpp)
target_compile_options(camera_hook_validation PRIVATE /UNDEBUG)
add_test(NAME camera_hook_validation COMMAND camera_hook_validation)
add_executable(camera_timing tests/timing.cpp)
add_test(NAME camera_timing COMMAND camera_timing)
add_executable(camera_body_position tests/body_position.cpp)
target_link_libraries(camera_body_position PRIVATE CommonLibSSE)
target_compile_options(camera_body_position PRIVATE /UNDEBUG)
add_test(NAME camera_body_position COMMAND camera_body_position)
